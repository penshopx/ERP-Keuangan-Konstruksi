### 3. Tax Engine Class (WAJIB)

```typescript
// lib/api-server/src/lib/tax-engine.ts

export class TaxEngine {
  
  /**
   * Hitung pajak dengan snapshot aturan pada tanggal tertentu
   * CRITICAL: Harus deterministic - input sama = output sama selamanya
   */
  async calculateInvoiceTaxes(params: {
    invoiceDate: Date;
    dpp: number;
    contractor: {
      businessType: 'PELAKSANA' | 'PERENCANA' | 'PENGAWAS';
      qualification: 'UMKM' | 'NON_UMKM';
      isPKP: boolean;
      sbuValid: boolean;
    };
    project: {
      isIntegrated: boolean;
      retentionPercent: number;
      collectorType: 'SELF' | 'GOVERNMENT_TREASURER';
    };
  }): Promise<TaxCalculationResult> {
    
    // 1. Ambil aturan yang berlaku pada invoiceDate
    const pphRule = await this.getEffectiveRule(
      'PPH_FINAL_4_2',
      `${params.contractor.businessType}_${params.contractor.qualification}`,
      params.invoiceDate
    );
    
    const ppnRule = await this.getEffectiveRule(
      'PPN_DPP_LAIN',
      params.project.isIntegrated ? 'TERINTEGRASI' : 'NON_TERINTEGRASI',
      params.invoiceDate
    );
    
    const ppnRateRule = await this.getEffectiveRule(
      'PPN_RATE', 'STANDARD', params.invoiceDate
    );
    
    // 2. Validasi pre-conditions
    if (!params.contractor.sbuValid) {
      throw new TaxValidationError(
        'Kontraktor tidak memiliki SBU valid. Tidak bisa apply PPh Final 4(2).'
      );
    }
    
    // 3. Hitung
    const pphFinal = params.dpp * pphRule.rate;
    
    let ppn = 0;
    if (params.contractor.isPKP) {
      const dppLain = params.dpp * ppnRule.rate;
      ppn = dppLain * ppnRateRule.rate;
    }
    
    // Jika pemungut pemerintah, PPN tidak dipungut oleh kontraktor
    if (params.project.collectorType === 'GOVERNMENT_TREASURER') {
      ppn = 0; // Akan dipungut langsung oleh bendaharawan
    }
    
    const retention = params.dpp * (params.project.retentionPercent / 100);
    const grandTotal = params.dpp + ppn - pphFinal - retention;
    
    // 4. Return dengan snapshot lengkap
    return {
      dpp: params.dpp,
      pphFinal,
      pphFinalRate: pphRule.rate,
      ppn,
      ppnRate: ppnRateRule.rate,
      dppLainRate: ppnRule.rate,
      retention,
      retentionRate: params.project.retentionPercent / 100,
      grandTotal,
      snapshot: {
        appliedAt: new Date().toISOString(),
        rulesVersion: {
          pphFinal: { 
            ruleId: pphRule.id, 
            rate: pphRule.rate, 
            regulation: pphRule.regulationSource 
          },
          ppn: { 
            ruleId: ppnRule.id, 
            dppLainRate: ppnRule.rate,
            ppnRate: ppnRateRule.rate,
            regulation: ppnRule.regulationSource 
          },
        },
        contractorQualification: params.contractor.qualification,
        businessType: params.contractor.businessType,
        isIntegrated: params.project.isIntegrated,
        isPKP: params.contractor.isPKP,
        collectorType: params.project.collectorType,
      }
    };
  }
  
  /**
   * Get aturan pajak yang berlaku pada tanggal tertentu
   * Logic: cari rule dengan effectiveDate <= targetDate 
   *        dan (expiryDate is null OR expiryDate > targetDate)
   */
  private async getEffectiveRule(
    ruleType: string, 
    ruleKey: string, 
    targetDate: Date
  ): Promise<TaxRule> {
    const rule = await db.query.taxRules.findFirst({
      where: and(
        eq(taxRules.ruleType, ruleType),
        eq(taxRules.ruleKey, ruleKey),
        lte(taxRules.effectiveDate, targetDate),
        or(
          isNull(taxRules.expiryDate),
          gt(taxRules.expiryDate, targetDate)
        )
      ),
      orderBy: desc(taxRules.effectiveDate),
    });
    
    if (!rule) {
      throw new TaxRuleNotFoundError(
        `No tax rule found for ${ruleType}:${ruleKey} on ${targetDate}`
      );
    }
    
    return rule;
  }
}
```

### 6. Edge Cases yang HARUS Di-handle

| Edge Case | Expected Behavior |
|-----------|-------------------|
| Kontraktor punya multi-SBU | Pilih SBU yang paling sesuai dengan scope pekerjaan di invoice |
| Kontraktor naik kualifikasi di tengah proyek | Invoice baru pakai tarif baru, invoice lama tetap tarif lama |
| Aturan pajak belum ada untuk tanggal tertentu | Throw error dengan message jelas, jangan fallback ke aturan terbaru |
| Invoice di-reversal | Buat invoice reversal baru, jangan edit invoice lama |
| Kontraktor asing (WPLN) | Apply PPh 26 (20%) atau P3B treaty jika ada |
| Konstruksi di luar negeri | Tidak apply PPh 4(2), tapi mungkin pajak negara setempat |
```

## AREA 1 — REVENUE RECOGNITION ENGINE (PSAK 72 / IFRS 15)

Ini adalah **jantung akuntansi konstruksi**. Salah implementasi = laporan keuangan salah = audit finding = sanksi OJK (untuk perusahaan publik).

### 1.1. Konsep Bisnis — Percentage of Completion (POC)

Kontrak konstruksi jangka panjang **tidak boleh** mengakui pendapatan sekaligus di akhir proyek. Harus diakui **bertahap** berdasarkan progres.

**Dua metode yang diizinkan:**

```
┌─────────────────────────────────────────────────────────────────┐
│ METODE                    │ RUMUS                │ KAPAN DIPAKAI │
├─────────────────────────────────────────────────────────────────┤
│ Cost-to-Cost Method       │ POC = Actual Cost /  │ Paling umum   │
│ (Input Method)            │         Total Budget │ di konstruksi │
├─────────────────────────────────────────────────────────────────┤
│ Milestone Method          │ POC = berdasarkan    │ Jika ada      │
│ (Output Method)           │ milestone tercapai   │ milestone     │
│                           │                      │ jelas         │
└─────────────────────────────────────────────────────────────────┘
```

### 1.3. Logika Implementasi — Revenue Recognition Engine

```typescript
class RevenueRecognitionEngine {
    
    /**
     * Hitung pengakuan pendapatan untuk satu periode
     */
    static calculatePeriodRevenue(
        contract: Contract,
        periodData: PeriodInput
    ): PeriodOutput {
        
        // 1. Hitung Percentage of Completion (POC)
        let pctComplete: number;
        
        if (contract.recognition_method === 'COST_TO_COST') {
            // POC = Biaya Aktual / Total Budget
            const totalBudget = periodData.revised_total_budget || contract.total_estimated_cost;
            pctComplete = periodData.actual_cost_to_date / totalBudget;
            
            // Validasi: POC tidak boleh > 100%
            if (pctComplete > 1.0) {
                throw new Error('POC exceeds 100%. Check cost data.');
            }
        } else if (contract.recognition_method === 'MILESTONE') {
            // POC berdasarkan milestone yang tercapai
            pctComplete = this.calculateMilestoneProgress(contract, periodData);
        }
        
        // 2. Hitung Revenue Kumulatif
        const cumulativeRevenue = contract.contract_value * pctComplete;
        
        // 3. Hitung Revenue Periode Ini
        const revenueRecognizedToDate = this.getRevenueRecognizedToDate(
            contract.id, 
            periodData.period_year, 
            periodData.period_month
        );
        const revenueThisPeriod = cumulativeRevenue - revenueRecognizedToDate;
        
        // 4. Hitung Cost Periode Ini
        const costRecognizedToDate = this.getCostRecognizedToDate(
            contract.id,
            periodData.period_year,
            periodData.period_month
        );
        const costThisPeriod = periodData.actual_cost_to_date - costRecognizedToDate;
        
        // 5. Hitung Gross Profit
        const grossProfitThisPeriod = revenueThisPeriod - costThisPeriod;
        
        // 6. Cek Onerous Contract (Estimated Loss)
        const estimatedTotalCost = periodData.actual_cost_to_date + periodData.estimated_cost_to_complete;
        const isOnerous = estimatedTotalCost > contract.contract_value;
        
        if (isOnerous) {
            // Akui seluruh loss sekarang (tidak boleh ditunda)
            const totalEstimatedLoss = estimatedTotalCost - contract.contract_value;
            const lossRecognizedToDate = this.getLossRecognizedToDate(contract.id);
            const lossThisPeriod = totalEstimatedLoss - lossRecognizedToDate;
            
            return {
                pct_complete: pctComplete,
                cumulative_revenue: cumulativeRevenue,
                revenue_this_period: revenueThisPeriod,
                cost_this_period: costThisPeriod,
                gross_profit_this_period: grossProfitThisPeriod - lossThisPeriod,
                loss_accrual: lossThisPeriod,
                is_onerous: true
            };
        }
        
        return {
            pct_complete: pctComplete,
            cumulative_revenue: cumulativeRevenue,
            revenue_this_period: revenueThisPeriod,
            cost_this_period: costThisPeriod,
            gross_profit_this_period: grossProfitThisPeriod,
            loss_accrual: 0,
            is_onerous: false
        };
    }
    
    /**
     * Generate jurnal akuntansi otomatis
     */
    static generateAccountingEntries(period: PeriodOutput): AccountingEntry[] {
        const entries: AccountingEntry[] = [];
        
        // Jurnal pengakuan pendapatan
        entries.push({
            account_code: '4100', // Revenue - Construction
            debit: 0,
            credit: period.revenue_this_period,
            description: `Revenue recognition - ${period.period_year}-${period.period_month}`
        });
        
        entries.push({
            account_code: '1300', // Accounts Receivable - Unbilled
            debit: period.revenue_this_period,
            credit: 0,
            description: `Unbilled revenue - ${period.period_year}-${period.period_month}`
        });
        
        // Jurnal pengakuan biaya
        entries.push({
            account_code: '5100', // Cost of Construction
            debit: period.cost_this_period,
            credit: 0,
            description: `Cost recognition - ${period.period_year}-${period.period_month}`
        });
        
        entries.push({
            account_code: '2400', // Construction in Progress (CIP)
            debit: 0,
            credit: period.cost_this_period,
            description: `CIP reduction - ${period.period_year}-${period.period_month}`
        });
        
        // Jika ada loss accrual
        if (period.loss_accrual > 0) {
            entries.push({
                account_code: '5900', // Loss on Construction Contracts
                debit: period.loss_accrual,
                credit: 0,
                description: `Loss accrual - ${period.period_year}-${period.period_month}`
            });
            
            entries.push({
                account_code: '2500', // Estimated Loss Provision
                debit: 0,
                credit: period.loss_accrual,
                description: `Loss provision - ${period.period_year}-${period.period_month}`
            });
        }
        
        return entries;
    }
}
```

### 1.4. Contoh Perhitungan Nyata

```
Kontrak: Rp 10 Miliar
Total Budget: Rp 8 Miliar
Margin rencana: 20%

Bulan 1:
├─ Biaya aktual: Rp 2 Miliar
├─ POC = 2M / 8M = 25%
├─ Revenue kumulatif = 25% × 10M = Rp 2.5 Miliar
├─ Revenue bulan 1 = Rp 2.5 Miliar (karena belum ada revenue sebelumnya)
├─ Cost bulan 1 = Rp 2 Miliar
└─ Gross profit = Rp 500 Juta

Bulan 2:
├─ Biaya aktual kumulatif: Rp 5 Miliar
├─ POC = 5M / 8M = 62.5%
├─ Revenue kumulatif = 62.5% × 10M = Rp 6.25 Miliar
├─ Revenue bulan 2 = 6.25M - 2.5M = Rp 3.75 Miliar
├─ Cost bulan 2 = 5M - 2M = Rp 3 Miliar
└─ Gross profit = Rp 750 Juta

Bulan 3 (terjadi cost overrun):
├─ Biaya aktual kumulatif: Rp 8.5 Miliar
├─ Estimasi sisa biaya: Rp 1 Miliar
├─ Total estimasi biaya: 8.5M + 1M = Rp 9.5 Miliar
├─ POC = 8.5M / 9.5M = 89.47%
├─ Revenue kumulatif = 89.47% × 10M = Rp 8.947 Miliar
├─ Revenue bulan 3 = 8.947M - 6.25M = Rp 2.697 Miliar
├─ Cost bulan 3 = 8.5M - 5M = Rp 3.5 Miliar
├─ Gross profit bulan 3 = 2.697M - 3.5M = -Rp 803 Juta (RUGI!)
└─ Margin total turun dari 20% menjadi 5.26%
```

### 1.5. Edge Cases Kritis

| Edge Case | Implementasi yang Benar |
|-----------|------------------------|
| **Perubahan estimasi biaya** | Hitung ulang POC secara **prospektif** (cumulative catch-up). Jangan reset history. |
| **Claim & dispute** | Jangan akui revenue dari claim sampai **probable** dan **measurable**. |
| **Variation Order (VO)** | Tambahkan ke contract value hanya jika **approved** oleh owner. |
| **Proyek dihentikan** | Akui revenue sampai tanggal terminasi. Hitung loss jika ada. |
| **Advance payment** | Bukan revenue. Akui sebagai **liability** sampai pekerjaan selesai. |

Ini area **paling rawan kebocoran**. Tanpa sistem yang ketat, Anda bisa bayar subkon **berlebihan** atau **terlambat** (kena denda).

### 2.1. Alur Lengkap Subkontraktor

```
[1] PRE-QUALIFICATION
    ├─ Verifikasi SBU, SIUP, NPWP
    ├─ Cek referensi proyek sebelumnya
    └─ Approval oleh Procurement Manager
        ↓
[2] TENDER & NEGOTIASI
    ├─ Undangan tender ke subkon ter-prequalifikasi
    ├─ Evaluasi teknis & komersial
    └─ Negosiasi harga & termin
        ↓
[3] KONTRAK SUBKON
    ├─ Scope of work detail
    ├─ Harga & termin pembayaran
    ├─ Retensi (biasanya 5%)
    ├─ Denda keterlambatan
    └─ Klausul backcharge
        ↓
[4] EKSEKUTI PEKERJAAN
    ├─ Subkon mobilisasi
    ├─ Daily report dari subkon
    └─ Quality control oleh Site Manager
        ↓
[5] OPNAME & TAGIHAN
    ├─ Subkon ajukan tagihan + bukti pekerjaan
    ├─ Site Manager opname volume aktual
    ├─ Bandingkan dengan kontrak
    └─ Hitung backcharge (jika ada)
        ↓
[6] PEMBAYARAN
    ├─ Hitung nilai bayar = Opname - Backcharge - Retensi - PPh 4(2)
    ├─ Generate Bupot PPh 4(2)
    └─ Transfer pembayaran
        ↓
[7] RETENSI & PEMELIHARAAN
    ├─ Retensi ditahan sampai masa pemeliharaan selesai
    ├─ Subkon perbaiki defect (jika ada)
    └─ Pencairan retensi setelah masa pemeliharaan
```

### 2.3. Logika Perhitungan Pembayaran Subkon

```typescript
class SubcontractorPaymentEngine {
    
    /**
     * Hitung nilai bayar ke subkon
     */
    static calculatePayment(
        opname: Opname,
        backcharges: Backcharge[],
        subcontract: Subcontract,
        subcontractor: Subcontractor
    ): PaymentCalculation {
        
        // 1. Nilai opname (volume aktual × harga satuan)
        const opnameAmount = opname.volume_actual * opname.unit_price;
        
        // 2. Total backcharge
        const totalBackcharge = backcharges
            .filter(bc => bc.status === 'APPROVED')
            .reduce((sum, bc) => sum + bc.amount, 0);
        
        // 3. Retensi
        const retentionAmount = opnameAmount * (subcontract.retention_percent / 100);
        
        // 4. Dasar Pengenaan Pajak (DPP)
        // DPP = Opname - Backcharge (backcharge mengurangi DPP)
        const taxableAmount = opnameAmount - totalBackcharge;
        
        // 5. PPh Final 4(2)
        const pphRate = this.getPphRate(subcontractor);
        const pphFinalAmount = taxableAmount * pphRate;
        
        // 6. Net Payable (Nilai bersih yang dibayar)
        // = Opname - Backcharge - Retensi - PPh Final
        const netPayable = opnameAmount - totalBackcharge - retentionAmount - pphFinalAmount;
        
        return {
            opname_amount: opnameAmount,
            backcharge_amount: totalBackcharge,
            retention_amount: retentionAmount,
            taxable_amount: taxableAmount,
            pph_final_amount: pphFinalAmount,
            pph_final_rate: pphRate,
            net_payable: netPayable
        };
    }
    
    /**
     * Hitung denda keterlambatan
     */
    static calculateDelayPenalty(
        subcontract: Subcontract,
        actualEndDate: Date
    ): DelayPenalty {
        const plannedEndDate = subcontract.end_date;
        
        if (actualEndDate <= plannedEndDate) {
            return {
                delay_days: 0,
                penalty_amount: 0
            };
        }
        
        const delayDays = this.daysBetween(plannedEndDate, actualEndDate);
        const penaltyPerDay = subcontract.contract_value * subcontract.penalty_per_day;
        const penaltyAmount = delayDays * penaltyPerDay;
        
        // Maksimal denda biasanya 5% dari nilai kontrak
        const maxPenalty = subcontract.contract_value * 0.05;
        const finalPenalty = Math.min(penaltyAmount, maxPenalty);
        
        return {
            delay_days: delayDays,
            penalty_amount: finalPenalty
        };
    }
    
    static getPphRate(subcontractor: Subcontractor): number {
        const rates = {
            PELAKSANA: { UMKM: 0.0175, NON_UMKM: 0.025 },
            PERENCANA: { UMKM: 0.035, NON_UMKM: 0.04 },
            PENGAWAS: { UMKM: 0.035, NON_UMKM: 0.04 }
        };
        
        const type = subcontractor.sbu_type;
        const qual = subcontractor.qualification === 'UMKM' ? 'UMKM' : 'NON_UMKM';
        
        return rates[type][qual];
    }
}
```

## AREA 3 — CASHFLOW FORECASTING & WORKING CAPITAL MANAGEMENT

Ini yang membedakan aplikasi **"biasa"** dengan **"enterprise-grade"**. Tanpa ini, perusahaan bisa **bangkrut** meskipun proyek untung.

### 3.1. Konsep Bisnis — Cashflow di Konstruksi

```
┌─────────────────────────────────────────────────────────────────┐
│ SUMBER CASH INFLOW              │ SUMBER CASH OUTFLOW           │
├─────────────────────────────────┼───────────────────────────────┤
│ Termin dari owner               │ Pembayaran ke subkon          │
│ Advance payment (DP)            │ Pembelian material            │
│ Pencairan retensi               │ Upah pekerja                  │
│ Klaim asuransi                  │ Sewa alat berat               │
│                                 │ Overhead proyek               │
│                                 │ Pembayaran pajak              │
│                                 │ Cicilan utang bank            │
└─────────────────────────────────────────────────────────────────┘
```

**Masalah klasik:**
- Owner bayar termin **telat** (60-90 hari)
- Subkon & vendor minta bayar **cepat** (14-30 hari)
- Pajak harus setor **bulan berikutnya**
- Retensi ditahan **6-12 bulan**

→ **Cashflow gap** bisa mencapai **miliaran rupiah**.

```sql
    accounts_payable DECIMAL(18,2), -- Utang usaha
    accruals DECIMAL(18,2), -- Biaya yang masih harus dibayar
    
    -- Perhitungan
    net_working_capital DECIMAL(18,2), -- (AR + Retention + Inventory) - (AP + Accruals)
    cash_required DECIMAL(18,2), -- Cash yang dibutuhkan untuk operasional
    
    created_at TIMESTAMP DEFAULT NOW()
);
```

### 3.3. Logika Forecasting Engine

```typescript
class CashflowForecastEngine {
    
    /**
     * Generate forecast arus kas untuk 12 bulan ke depan
     */
    static generateForecast(
        project: Project,
        schedule: ProjectSchedule,
        contracts: Contract[]
    ): CashflowForecast[] {
        const forecasts: CashflowForecast[] = [];
        const startDate = new Date();
        const endDate = new Date(startDate.getFullYear() + 1, startDate.getMonth());
        
        let cumulativeCashflow = 0;
        
        for (let date = startDate; date <= endDate; date.setMonth(date.getMonth() + 1)) {
            const year = date.getFullYear();
            const month = date.getMonth() + 1;
            
            // 1. Forecast Inflow
            const expectedTermin = this.forecastTerminInflow(project, year, month);
            const expectedRetentionRelease = this.forecastRetentionRelease(project, year, month);
            const expectedAdvancePayment = this.forecastAdvancePayment(project, year, month);
            const totalInflow = expectedTermin + expectedRetentionRelease + expectedAdvancePayment;
            
            // 2. Forecast Outflow
            const expectedSubcontrPayment = this.forecastSubcontrPayment(project, year, month);
            const expectedMaterialPayment = this.forecastMaterialPayment(project, year, month);
            const expectedLaborPayment = this.forecastLaborPayment(project, year, month);
            const expectedEquipmentPayment = this.forecastEquipmentPayment(project, year, month);
            const expectedOverheadPayment = this.forecastOverheadPayment(project, year, month);
            const expectedTaxPayment = this.forecastTaxPayment(project, year, month);
            const totalOutflow = expectedSubcontrPayment + expectedMaterialPayment + 
                                 expectedLaborPayment + expectedEquipmentPayment + 
                                 expectedOverheadPayment + expectedTaxPayment;
            
            // 3. Net Cashflow
            const netCashflow = totalInflow - totalOutflow;
            cumulativeCashflow += netCashflow;
            
            forecasts.push({
                project_id: project.id,
                forecast_year: year,
                forecast_month: month,
                expected_termin: expectedTermin,
                expected_retention_release: expectedRetentionRelease,
                expected_advance_payment: expectedAdvancePayment,
                total_inflow: totalInflow,
                expected_subcontr_payment: expectedSubcontrPayment,
                expected_material_payment: expectedMaterialPayment,
                expected_labor_payment: expectedLaborPayment,
                expected_equipment_payment: expectedEquipmentPayment,
                expected_overhead_payment: expectedOverheadPayment,
                expected_tax_payment: expectedTaxPayment,
                total_outflow: totalOutflow,
                net_cashflow: netCashflow,
                cumulative_cashflow: cumulativeCashflow
            });
        }
        
        return forecasts;
    }
    
    /**
     * Forecast termin inflow berdasarkan Kurva S
     */
    static forecastTerminInflow(
        project: Project,
        year: number,
        month: number
    ): number {
        // Ambil progres fisik yang direncanakan dari Kurva S
        const plannedProgress = this.getPlannedProgress(project.id, year, month);
        
        // Hitung nilai termin berdasarkan progres
        const terminValue = project.contract_value * plannedProgress;
        
        // Kurangi dengan termin yang sudah dibayar
        const paidTermin = this.getPaidTermin(project.id, year, month);
        
        // Kurangi retensi
        const retention = terminValue * (project.retention_percent / 100);
        
        // Kurangi PPh Final dan PPN (karena dipotong pemotong pajak)
        const pphFinal = terminValue * project.pph_final_rate;
        const ppn = terminValue * 0.11; // Atau DPP Lain
        
        // Net termin yang diterima
        const netTermin = terminValue - retention - pphFinal - ppn;
        
        // Apply payment delay (misal: owner bayar 60 hari setelah invoice)
        // Maka termin bulan ini baru diterima 2 bulan lagi
        const delayMonths = 2;
        const expectedMonth = month + delayMonths;
        
        return netTermin;
    }
    
    /**
     * Forecast pembayaran ke subkon
     */
    static forecastSubcontrPayment(
        project: Project,
        year: number,
        month: number
    ): number {
        // Ambil rencana kerja dari schedule
        const plannedWork = this.getPlannedWork(project.id, year, month);
        
        // Hitung nilai pekerjaan subkon berdasarkan rencana
        const subcontrValue = plannedWork.subcontr_value;
        
        // Kurangi retensi
        const retention = subcontrValue * 0.05;
        
        // Kurangi PPh Final
        const pphFinal = subcontrValue * 0.025; // Asumsi Non-UMKM
        
        // Net pembayaran
        const netPayment = subcontrValue - retention - pphFinal;
        
        // Apply payment terms (misal: bayar 30 hari setelah opname)
        return netPayment;
    }
    
    /**
     * Hitung working capital requirement
     */
    static calculateWorkingCapital(project: Project): WorkingCapital {
        // Accounts Receivable (Piutang)
        const accountsReceivable = this.getOutstandingInvoices(project.id);
        
        // Retention Receivable
        const retentionReceivable = this.getRetentionReceivable(project.id);
        
        // Inventory (Stok material)
        const inventory = this.getInventoryValue(project.id);
        
        // Accounts Payable (Utang)
        const accountsPayable = this.getOutstandingBills(project.id);
        
        // Accruals (Biaya yang masih harus dibayar)
        const accruals = this.getAccruedExpenses(project.id);
        
        // Net Working Capital
        const netWorkingCapital = (accountsReceivable + retentionReceivable + inventory) - 
                                  (accountsPayable + accruals);
        
        // Cash Required (minimal 3 bulan operasional)
        const monthlyBurnRate = this.calculateMonthlyBurnRate(project);
        const cashRequired = monthlyBurnRate * 3;
        
        return {
            accounts_receivable: accountsReceivable,
            retention_receivable: retentionReceivable,
            inventory: inventory,
            accounts_payable: accountsPayable,
            accruals: accruals,
            net_working_capital: netWorkingCapital,
            cash_required: cashRequired
        };
    }
}
```
